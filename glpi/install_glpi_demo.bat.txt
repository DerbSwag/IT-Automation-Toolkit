@echo off
setlocal

echo ==========================================
echo GLPI Agent Demo Installer
echo ==========================================

:: Load config
for /f "tokens=1,2 delims==" %%a in (config.ini) do (
    if "%%a"=="SERVER_URL" set SERVER_URL=%%b
    if "%%a"=="MSI_NAME" set MSI_NAME=%%b
)

echo Using server: %SERVER_URL%

:: Simulate install
echo [DEMO] Installing GLPI Agent...
timeout /t 2 >nul

echo [DEMO] Sending inventory to %SERVER_URL%
timeout /t 2 >nul

echo [SUCCESS] Demo completed
pause
